\newsection{Mengen, Relationen und Abbildungen}{task-1}

\begin{enumerate}
  \item {
    Es seien N, M zwei Mengen mit $|N| = n, |M| = m$ und $|N \setminus M| = d$.
    Geben Sie die folgenden Kardinalitäten an.
    Begründen Sie Ihre Antwort kurz, indem Sie die Zwischenschenschritte angeben.
    \begin{enumerate}
      \item {
        $|\mathcal{P}(M \cup N)| = $
        \solution{
          $2^{|(M \cup N)|} = 2^{|(M \cup (N \setminus M))|} = 2^{|M| + |(N \setminus M)|} = 2^{m + d}$
          \vspace*{3.5cm}
        }{\vspace*{4cm}}
      }
      \item {
        $|(M \cap N) \times M| = $
        \solution{
          $|(M \cap N)| \cdot |M| = |(N \setminus (N \setminus M))| \cdot |M| = (|N| - |(N \setminus M)|) \cdot |M| = (n - d) \cdot m$
          \vspace*{3.5cm}
        }{\vspace*{4cm}}
      }
    \end{enumerate}
  }
  \item {
    Zeigen Sie mittels vollständiger Induktion, dass für alle $n \in \mathbb{N}$
    gilt:
    \[
      \sum_{i=0}^{n-1} (3i^2 + 3i + 1) = n^3
    \]

    \solution{
      IA: Sei $n = 1 \in \mathbb{N}$
      \begin{flalign*}
        & \sum_{i=0}^{1-1} (3i^2 + 3i + 1) = \sum_{i=0}^{0} (3i^2 + 3i + 1) = (3 \cdot 0^2 + 3 \cdot 0 + 1) = 1 = 1^3 &
      \end{flalign*}
      \begin{flushright}
        \Checkmark
      \end{flushright}

      IV: Sei $n \in \mathbb{N}$ beliebig, aber fest. Dann gilt Aussage. \\

      IS:
      \begin{flalign*}
        \sum_{i=0}^{n} (3i^2 + 3i + 1) &= \sum_{i=0}^{n-1} (3i^2 + 3i + 1) + (3n^2 + 3n + 1) && \\
        &\overset{\text{IV}}{=} n^3 + 3n^2 + 3n + 1 \overset{\text{1. Binomische Formel}}{=} (n + 1)^3 &&
        \Box
      \end{flalign*}
    }{}
  }

  \item {

    \textit{surjektiv} oder \textit{weder noch} ist.
    Eine Begründung ist \textit{nicht} notwendig.

    \setlength{\tabcolsep}{10pt}
    \renewcommand{\arraystretch}{2}
    \begin{tabular}{@{}l c c c@{}}
      & $f$ injektiv & $f$ surjektiv & $f$ weder injektiv noch surjektiv \\
      $f: \mathbb{R} \to \mathbb{R} $ mit $f(x) = x^2 - 2x$ & $\square$ & $\square$ & $\correctsquare$ \\
      $f: \mathcal{P}(\mathbb{N}) \to \mathbb{N}_0 $ mit $f(A) = |A|$ & $\square$ & $\correctsquare$ & $\square$ \\[10pt]
      $f: \mathbb{R} \to \mathbb{R} $ mit $f(x) =
        \begin{cases}

        \end{cases}
        $ & $\correctsquare$ & $\correctsquare$ & $\square$ \\[10pt]
      $f: \mathbb{R}^2 \to \mathbb{R}^2 $ mit $f(x, y) = (y-7, \frac{1}{4} x)$ & $\correctsquare$ & $\correctsquare$ & $\square$ \\
      $f: \mathbb{R}^2_{\geq 0} \to \mathbb{R}_{\geq 0} $ mit $f(x, y) = x^y$ & $\square$ & $\correctsquare$ & $\square$ \\

  \item {
    Seien $M = \{A, B, C, D\}$ eine Menge und $R \subseteq M \times M$ die unten
    schematisch dargestellte Relation auf $M$. Geben Sie jeweils mit
    \textit{kurzer} Begründung an, ob $R$ reflexiv, symmetrisch, transitiv,
    antisymmetrisch bzw. eine Äquivalenzrelation ist.

    \begin{center}
    \begin{tikzpicture}[node distance=2cm, auto]
      \node[state] (A)  {$A$};
      \node[state] (B) [right=of A]  {$B$};
      \node[state] (C) [below=of A]  {$C$};
      \node[state] (D) [below=of B]  {$D$};
      \path[->]
      (A) edge [bend left =15] node {} (B)
      (B) edge [bend right =-15] node {} (A)
      (A) edge [bend left =15] node {} (C)
      (C) edge [bend right =-15] node {} (A)
      (B) edge [bend left =15] node {} (D)
      (D) edge [bend right =-15] node {} (B)
      (B) edge [bend left =15] node {} (C)
      (C) edge [bend right =-15] node {} (B);
    \end{tikzpicture}
    \end{center}
    \vspace{\fill}

    \begin{minipage}{\linewidth}
      reflexiv: \solution{
        Nein, da keine Schlingen vorhanden sind. Z.B. $A\cancel{R}A$.
      }{\hrulefill} \\[10pt]
      symmetrisch: \solution{
        Ja, da zu jeder Verbindung auch eine Verbindung in die entgegengesetzte
        Richtung existiert.
        Z.B. $ARB$ und $BRA$ oder $CRB$ und $BCR$.
        \\[10pt]
      }{\hrulefill \\[10pt] \hrule \vspace{18pt}}
      transitiv: \solution{
        Nein, da z.B. $ARB$ und $BRD$, aber $A\cancel{R}D$.
        \\[10pt]
      }{\hrulefill \\[10pt] \hrule \vspace{18pt}}
      antisymmetrisch: \solution{
        Nein, da symmetrisch und z.B. $ARB$ und $BRA$, aber $A \neq B$.
        \\[10pt]
      }{\hrulefill \\[10pt] \hrule \vspace{18pt}}
      Äquivalenzrelation: \solution{
        Nein, da nicht reflexiv und nicht transitiv.
      }{\hrulefill}
    \end{minipage}
  }
  \vspace{\fill}
\end{enumerate}
